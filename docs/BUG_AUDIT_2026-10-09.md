## Read-only checks to run in production

```sql
SELECT tablename, policyname, permissive, cmd, roles, qual FROM pg_policies
WHERE schemaname='public' AND tablename IN ('profiles','activity_logs','attendance_logs','location_tracking','field_visits','leave_requests','tasks','audit_logs','notifications')
ORDER BY tablename, policyname;

SELECT t.tgname, t.tgfoid::regprocedure, p.prosecdef FROM pg_trigger t JOIN pg_proc p ON p.oid=t.tgfoid
WHERE t.tgrelid='public.tasks'::regclass AND NOT t.tgisinternal;

SELECT oid::regprocedure, prosecdef FROM pg_proc
WHERE prosrc ILIKE '%audit_logs%' AND prosrc NOT ILIKE '%attendance_audit_logs%';

SELECT o.id, o.name, o.is_active,
  (SELECT count(*) FROM profiles p WHERE p.office_id=o.id) employees,
  (SELECT count(*) FROM user_permissions u WHERE u.scope_id=o.id::text AND u.revoked_at IS NULL) grants
FROM offices o WHERE o.name ILIKE '%alpha%' OR o.name ILIKE '%greater noida%';

SELECT id, public, file_size_limit, allowed_mime_types FROM storage.buckets WHERE id LIKE 'event%';
SELECT version FROM supabase_migrations.schema_migrations WHERE version LIKE '2026100815%';
SELECT column_name FROM information_schema.columns WHERE table_name='profiles' AND column_name='home_address';
```
